cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2_w CACHE STRING "Board type")
option(EOS_SIMULATOR_IMAGE "Emit a no_flash image for pico-sim" OFF)
if(EOS_SIMULATOR_IMAGE)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "" FORCE)
    set(CMAKE_C_FLAGS_DEBUG "-O0 -g" CACHE STRING "" FORCE)
    set(CMAKE_CXX_FLAGS_DEBUG "-O0 -g" CACHE STRING "" FORCE)
endif()
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)

if(NOT DEFINED ENV{FREERTOS_KERNEL_PATH})
    message(FATAL_ERROR "Set FREERTOS_KERNEL_PATH to Raspberry Pi's FreeRTOS-Kernel checkout")
endif()
set(_kernel "$ENV{FREERTOS_KERNEL_PATH}")
set(_imports
    "${_kernel}/portable/ThirdParty/GCC/RP2350_ARM_NTZ/FreeRTOS_Kernel_import.cmake"
    "${_kernel}/portable/ThirdParty/Community-Supported-Ports/GCC/RP2350_ARM_NTZ/FreeRTOS_Kernel_import.cmake")
foreach(_candidate IN LISTS _imports)
    if(EXISTS "${_candidate}")
        set(_freertos_import "${_candidate}")
        break()
    endif()
endforeach()
if(NOT _freertos_import)
    message(FATAL_ERROR "RP2350 FreeRTOS import file not found under ${_kernel}")
endif()

project(eos_l1_equal_priority C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
include("${_freertos_import}")

add_executable(l1_equal_priority main.c ../01-task-baseline/rtos_hooks.c)
target_include_directories(l1_equal_priority PRIVATE . ../01-task-baseline)
target_link_libraries(l1_equal_priority pico_stdlib FreeRTOS-Kernel-Heap4)
if(EOS_SIMULATOR_IMAGE)
    # The course simulator requires -O0 for the FreeRTOS kernel.
    target_compile_options(l1_equal_priority PRIVATE -O0)
endif()
option(EOS_EXPLICIT_YIELD "Have each task call taskYIELD after toggling" OFF)
if(EOS_EXPLICIT_YIELD)
    target_compile_definitions(l1_equal_priority PRIVATE EOS_EXPLICIT_YIELD=1)
else()
    target_compile_definitions(l1_equal_priority PRIVATE EOS_EXPLICIT_YIELD=0)
endif()
if(EOS_SIMULATOR_IMAGE)
    pico_set_binary_type(l1_equal_priority no_flash)
endif()
pico_enable_stdio_usb(l1_equal_priority 1)
pico_enable_stdio_uart(l1_equal_priority 1)
pico_add_extra_outputs(l1_equal_priority)
